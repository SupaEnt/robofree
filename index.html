import tkinter as tk import ctypes from ctypes import wintypes # === Windows API Constants (required for SetWindowPos) === SWP_NOMOVE = 0x0002 SWP_NOSIZE = 0x0001 SWP_NOACTIVATE = 0x0010 def create_floating_box(): root = tk.Tk() # Window setup root.title("Barak Laniado Enterprises") root.overrideredirect(True) # Frameless window root.attributes('-topmost', True) root.geometry("903x90+150+150") # Wide elegant banner # === Stronger Topmost for Browsers (Chrome, Edge, Firefox, etc.) === def force_topmost(): """Force the window to stay truly on top, even over browser UI elements.""" try: hwnd = root.winfo_id() ctypes.windll.user32.SetWindowPos( hwnd, wintypes.HWND(-1), # HWND_TOPMOST 0, 0, 0, 0, SWP_NOMOVE | SWP_NOSIZE | SWP_NOACTIVATE ) except Exception: pass # Apply immediately force_topmost() # Elegant dark theme with gold accents root.configure(bg='#0d0d1a') # Main frame with rich golden border frame = tk.Frame( root, bg='#0d0d1a', highlightbackground='#D4AF37', highlightthickness=2, bd=0 ) frame.pack(fill='both', expand=True, padx=4, pady=4) # Content container content_frame = tk.Frame(frame, bg='#0d0d1a') content_frame.pack(expand=True, pady=10) # Company name name_label = tk.Label( content_frame, text="Barak Laniado Enterprises", font=("Segoe UI", 20, "bold"), fg="#D4AF37", bg='#0d0d1a' ) name_label.pack(side='left', padx=(0, 15)) # 5-star rating stars_label = tk.Label( content_frame, text="★★★★★", font=("Segoe UI", 18), fg="#D4AF37", bg='#0d0d1a' ) stars_label.pack(side='left') # === Dragging support === root._drag_x = 0 root._drag_y = 0 def start_move(event): root._drag_x = event.x_root root._drag_y = event.y_root def do_move(event): deltax = event.x_root - root._drag_x deltay = event.y_root - root._drag_y new_x = root.winfo_x() + deltax new_y = root.winfo_y() + deltay root.geometry(f"+{new_x}+{new_y}") root._drag_x = event.x_root root._drag_y = event.y_root force_topmost() # Re-assert topmost after move root.bind("", start_move) root.bind("", do_move) # === Close button === def close_window(event=None): root.destroy() close_btn = tk.Button( frame, text="✕", command=close_window, bg='#0d0d1a', fg='#D4AF37', activebackground='#1a1a2e', activeforeground='#FFFFFF', bd=0, font=("Segoe UI", 11, "bold"), cursor="hand2", width=2, height=1 ) close_btn.place(relx=0.97, rely=0.08, anchor='ne') # Keyboard + mouse close shortcuts root.bind("", close_window) root.bind("", close_window) root.minsize(300, 70) root.mainloop() if __name__ == "__main__": create_floating_box()
